Private Sub Command1_Click()
Dim i As Integer, s As String
Dim c As Integer, cc As Integer
c = 0
cc = 0
Text2 = """"
For i = 1 To Len(Text1)
    s = Mid(Text1, i, 1)
    If Asc(s) >= 0 Then
        If Asc(s) = Asc(vbCrLf) Then
            Text2 = Text2 + """," + vbCrLf + """"
            cc = cc + 1
            c = 0
        Else
            c = c + 6
            If c >= 128 Then
                c = 6
                Text2 = Text2 + "\n"
            End If
            Text2 = Text2 + s
        End If
    Else
        c = c + 12
        If c >= 128 Then
            c = 12
            Text2 = Text2 + "\n"
        End If
        Text2 = Text2 + "\x" + Hex(&HFF + (Asc(s) \ &H100)) + "\x" + Hex(Asc(s) And &HFF)
    End If
Next i
Text2 = Text2 + """"
Me.Caption = cc + 1
End Sub
