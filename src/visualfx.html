static is_multichar(char c)
{
	return c&0x80;

static void puttext(int x,int y,vf_FONT* font,int width,int height,int sel,char* text)
{
	int pt,pb,c,lim,yr;
	char printbuff[255];

	lim=(width+1)/(font->width+1);
	pt=0;yr=0;
	while(text[pt] && yr+font->height<=height){
		pb=0;c=0;
		while(text[pt] && c<lim){
			if(text[pt]=='\n'){pt++;break;}
			if(is_multichar(printbuff[pb++]=text[pt++])) printbuff[pb++]=text[pt++];
			c++;
		}
		printbuff[pb]=0;
		font->putstring(x,y+yr,sel,printbuff);
		yr+=font->height+1;
	}
}

#include <stdio.h>
#include <fxlib.h>

#include "fxchslib.h"

#define is_chschar(c) ((c) & 0x80)


typedef FONTCHARACTER fontc;

FONTFILE * def_font = 0;

static void draw_picture (int x,int y,int width,int height,int mode,unsigned char * pGraph)
{
	DISPGRAPH dg;

	dg.x					= x;
	dg.y					= y;
	dg.GraphData.width		= width;
	dg.GraphData.height		= height;
	dg.GraphData.pBitmap	= pGraph;
	dg.WriteModify			= mode+1;
	dg.WriteKind			= IMB_WRITEKIND_OVER;

	Bdisp_WriteGraph_VRAM(&dg);
}

static unsigned int x86_dword_to_sh (unsigned int d)
{
	unsigned char c1,c2,c3,c4;
	unsigned int  r;

	c1 = d>>24	;
	c2 = (d & 0xFF0000)		>>16;
	c3 = (d & 0xFF00)		>>8	;
	c4 = (d & 0xFF)				;
	
	r = c1 + (c2<<8) + (c3<<16) + (c4<<24);

	return r;
}

static fontc * char_to_font(const char *cfname,fontc *ffname)
{
	int i,len = strlen(cfname);

	for(i=0; i<len ;++i)
		ffname[i] = cfname[i];

	ffname[i]=0;
	return ffname;
}

FONTFILE * open_font(const char * cfname)
{
	int			fh,r;
	FONTFILE	*ff;
	fontc		ffname[32];

	char_to_font(cfname,ffname);

	fh = Bfile_OpenFile(ffname,_OPENMODE_READ_SHARE);

	if (fh<=0) return NULL;

	ff			 		= (FONTFILE*)malloc(sizeof(FONTFILE));
	Bfile_ReadFile (fh,ff,sizeof(FONTFILE)-2*sizeof(int),0);

	ff->file_handle		= fh;

	ff->width			= x86_dword_to_sh(ff->width);
	ff->height			= x86_dword_to_sh(ff->height);	
	ff->asc_offset		= x86_dword_to_sh(ff->asc_offset);
	ff->chs_offset		= x86_dword_to_sh(ff->chs_offset);

	r					=ff->width/8 + (ff->width % 8 ? 1 : 0);

	ff->font_size		= r*ff->height;

	return ff;
}

void close_font (FONTFILE * ff)
{
	Bfile_CloseFile(ff->file_handle);
	free(ff);
}


void select_font (const FONTFILE * font)
{
	def_font = font;
}

void print_chs_char (int x,int y,int sel,unsigned char c1,unsigned char c2)
{
	unsigned char mat[128];
	int sec,pot;

	if (def_font==NULL) return;

 
	sec = c1-0xa1;
	pot = c2-0xa1;

	Bfile_ReadFile( def_font->file_handle,
					mat,
					def_font->font_size,
					(94*sec+pot)*def_font->font_size + def_font->chs_offset);

	draw_picture(x,y,def_font->width,def_font->height,sel,mat);
}

void print_asc_char (int x,int y,int sel,unsigned char c)
{
	unsigned char mat[128];

	if (def_font==NULL) return;


	Bfile_ReadFile( def_font->file_handle,
					mat,
					def_font->font_size,
					c*def_font->font_size + def_font->asc_offset);

	draw_picture(x,y,def_font->width,def_font->height,sel,mat);
}

void print_chs_str (int x,int y,int sel,const unsigned char * str)
{
	int		i;
	int		l = 0;
	int		cx;

	if (! def_font) return ;

	for (i=0;str[i];l++)
	{
		cx = x+l*def_font->width;

		if (is_chschar(str[i]))
		{
			
			print_chs_char (cx,y,sel,str[i],str[i+1]);
			++i,++i;
		}
		else
		{
			++i;
			if (! def_font->asc_offset) continue;
			print_asc_char (cx,y,sel,str[i-1]);
		}
		
	}
}

#define MAX_CONTROL_NUMBER 50
#define MINI_WIDTH 3
#define MINI_HEIGHT 5
#define BIG_WIDTH 5
#define BIG_HEIGHT 7
#define MAX_WIDTH 128
#define MAX_PRINT_BUFFER 33
//MAX_WIDTH/(MINI_WIDTH+1)+1

typedef enum{NULL_TYPE,CAPTION,PASSAGE} vf_CONTROLTYPE;
typedef struct{
	int width;
	int height;
	void (*printptr)(int x,int y,char* str); 
} vf_FONT;
typedef struct{
	vf_CONTROLTYPE type;
	int x;int y;int width;int height;
	int font;
	char *text;
} vf_CONTROL;

vf_CONTROL vf_control[MAX_CONTROL_NUMBER];

void vf_refresh()
{
	int i,j,k,l,x,y,yr,w,h;
	char printbuff[MAX_PRINT_BUFFER];

	Bdisp_AllClr_VRAM();
	for(i=0;i<MAX_CONTROL_NUMBER;i+=1){
		x=vf_control[i].x;y=vf_control[i].y;
		w=vf_control[i].width;h=vf_control[i].height;
		switch(vf_control[i].type){
			case CAPTION:
				l=(vf_control[i].width+1)/(BIG_WIDTH+1);
				j=0;
				while(vf_control[i].text[j] && j<l){printbuff[j]=vf_control[i].text[j];j++;}
				printbuff[j]=0;
				PrintXY(x,y,printbuff,0);
			break;
			case PASSAGE:
				l=(vf_control[i].width+1)/(MINI_WIDTH+1);
				j=0;yr=0;
				while(vf_control[i].text[j] && yr+MINI_HEIGHT<=vf_control[i].height){
					k=0;
					while(vf_control[i].text[j] && k<l){printbuff[k]=vf_control[i].text[j];j++;k++;}
					printbuff[k]=0;
					PrintMini(x,y+yr,printbuff,0);
					yr+=MINI_HEIGHT+1;
				}
			break;
		}//switch
	}//for
	Bdisp_PutDisp_DD();
}
int vf_create()
{
	int i;
	for(i=0;i<MAX_CONTROL_NUMBER;i+=1)
		if(vf_control[i].type==NULL_TYPE) return i;
	return -1;
}
int vf_addcaption(int x,int y,int width,char *text)
{
	int i;
	if((i=vf_create())==-1) return -1;
	vf_control[i].type=CAPTION;
	vf_control[i].x=x;vf_control[i].y=y;vf_control[i].width=width;vf_control[i].text=text;
	vf_control[i].height=BIG_HEIGHT;
	return i;
}
int vf_addpassage(int x,int y,int width,int height,char *text)
{
	int i;
	if((i=vf_create())==-1) return -1;
	vf_control[i].type=PASSAGE;
	vf_control[i].x=x;vf_control[i].y=y;vf_control[i].width=width;vf_control[i].height=height;vf_control[i].text=text;
	return i;
}

#include <stdio.h>
#include <fxlib.h>

#include "fxchslib.h"

#define is_chschar(c) ((c) & 0x80)


typedef FONTCHARACTER fontc;

FONTFILE * def_font = 0;

static void draw_picture (int x,int y,int width,int height,int mode,unsigned char * pGraph)
{
	DISPGRAPH dg;

	dg.x					= x;
	dg.y					= y;
	dg.GraphData.width		= width;
	dg.GraphData.height		= height;
	dg.GraphData.pBitmap	= pGraph;
	dg.WriteModify			= mode+1;
	dg.WriteKind			= IMB_WRITEKIND_OVER;

	Bdisp_WriteGraph_VRAM(&dg);
}

static unsigned int x86_dword_to_sh (unsigned int d)
{
	unsigned char c1,c2,c3,c4;
	unsigned int  r;

	c1 = d>>24	;
	c2 = (d & 0xFF0000)		>>16;
	c3 = (d & 0xFF00)		>>8	;
	c4 = (d & 0xFF)				;
	
	r = c1 + (c2<<8) + (c3<<16) + (c4<<24);

	return r;
}

static fontc * char_to_font(const char *cfname,fontc *ffname)
{
	int i,len = strlen(cfname);

	for(i=0; i<len ;++i)
		ffname[i] = cfname[i];

	ffname[i]=0;
	return ffname;
}

FONTFILE * open_font(const char * cfname)
{
	int			fh,r;
	FONTFILE	*ff;
	fontc		ffname[32];

	char_to_font(cfname,ffname);

	fh = Bfile_OpenFile(ffname,_OPENMODE_READ_SHARE);

	if (fh<=0) return NULL;

	ff			 		= (FONTFILE*)malloc(sizeof(FONTFILE));
	Bfile_ReadFile (fh,ff,sizeof(FONTFILE)-2*sizeof(int),0);

	ff->file_handle		= fh;

	ff->width			= x86_dword_to_sh(ff->width);
	ff->height			= x86_dword_to_sh(ff->height);	
	ff->asc_offset		= x86_dword_to_sh(ff->asc_offset);
	ff->chs_offset		= x86_dword_to_sh(ff->chs_offset);

	r					=ff->width/8 + (ff->width % 8 ? 1 : 0);

	ff->font_size		= r*ff->height;

	return ff;
}

void close_font (FONTFILE * ff)
{
	Bfile_CloseFile(ff->file_handle);
	free(ff);
}


void select_font (const FONTFILE * font)
{
	def_font = font;
}

void print_chs_char (int x,int y,int sel,unsigned char c1,unsigned char c2)
{
	unsigned char mat[128];
	int sec,pot;

	if (def_font==NULL) return;

 
	sec = c1-0xa1;
	pot = c2-0xa1;

	Bfile_ReadFile( def_font->file_handle,
					mat,
					def_font->font_size,
					(94*sec+pot)*def_font->font_size + def_font->chs_offset);

	draw_picture(x,y,def_font->width,def_font->height,sel,mat);
}

void print_asc_char (int x,int y,int sel,unsigned char c)
{
	unsigned char mat[128];

	if (def_font==NULL) return;


	Bfile_ReadFile( def_font->file_handle,
					mat,
					def_font->font_size,
					c*def_font->font_size + def_font->asc_offset);

	draw_picture(x,y,def_font->width,def_font->height,sel,mat);
}

void print_chs_str (int x,int y,int sel,const unsigned char * str)
{
	int		i;
	int		l = 0;
	int		cx;

	if (! def_font) return ;

	for (i=0;str[i];l++)
	{
		cx = x+l*def_font->width;

		if (is_chschar(str[i]))
		{
			
			print_chs_char (cx,y,sel,str[i],str[i+1]);
			++i,++i;
		}
		else
		{
			++i;
			if (! def_font->asc_offset) continue;
			print_asc_char (cx,y,sel,str[i-1]);
		}
		
	}
}
