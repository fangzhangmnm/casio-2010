int db_init_player()
{
	int fp;
	PLAYER player;
	int i,p,n;

	if((fp=Bfile_OpenMainMemory("PLAYER"))<0) return 1;
	i=0;p=0;n=0;
	while(Bfile_ReadFile(fp,&player,sizeof(PLAYER),-1)>=0 && i<MAX_PLAYER){
		if(player.avilable){
			playerkey[i].id=player.id;
			strcpy(playerkey[i].username,player.username);
			playerkey[i].position=p;
			n++;
		}
		p++;
	}
	Bfile_CloseFile(fp);
	if(global.playernum!=n) return 1;//alert
	return 0;
}
int db_search_player(char* username)
{
	int i;
	for(i=0;i<global.playernum;i++)
		if(strcmp(username,playerkey[i].username)) return playerkey[i].id;
	return -1;
}
int db_load_player(int id,PLAYER* buff)
{
	int i,p;
	int fp;
	p=-1;
	for(i=0;i<global.playernum;i++)
		if(playerkey[i].id==id) p=i;
	if(p==-1) return 1;
	if((fp=Bfile_OpenMainMemory("PLAYER"))<0) return 1;
	Bfile_SeekFile(fp,p*sizeof(PLAYER));
	if(Bfile_ReadFile(fp,buff,sizeof(PLAYER),-1)<0) return 1;
	return 0;
}

#define INIT_YEAR 2012

#define nSecondsInJan 2678400L
#define nSecondsInApril 2592000L
#define nSecondsInLeapFeb 2505600L
#define nSecondsInNonLeapFeb 2419200L
const nSecondsInNonLeapYear=31536000L;
const nSecondsInLeapYear=31622400L;
const long int nSecondsInMonths_NonLeap[] ={
	0,nSecondsInJan, nSecondsInNonLeapFeb, nSecondsInJan, nSecondsInApril,nSecondsInJan, nSecondsInApril, nSecondsInJan, nSecondsInJan,nSecondsInApril, nSecondsInJan, nSecondsInApril, nSecondsInJan};
const long int nSecondsInMonths_Leap[] ={
	0,nSecondsInJan, nSecondsInLeapFeb, nSecondsInJan, nSecondsInApril,nSecondsInJan, nSecondsInApril, nSecondsInJan, nSecondsInJan,nSecondsInApril, nSecondsInJan, nSecondsInApril, nSecondsInJan};

typedef struct{
	short int year;//INIT_YEAR-INIT_YEAR+67
	short int month;//1-12
	short int day;//1-31
	short int hour;
	short int minute;
	short int second;
} time;
typedef enum{
	DATE_AND_TIME,TIME,COUNTDOWN
} TIMEFORMAT;

int isleapyear(int n)
{
	if(n%100==0) return (n%400==0); else return (n%4==0);
}
long int getsecond(time t)
{
	int year,month;
	long int n = 0;
	for (year = INIT_YEAR; year < t.year; year++){
		if (isleapyear(year)) n+=nSecondsInLeapYear;
		else n+=nSecondsInNonLeapYear;
	}
	for (month = 1; month < t.month; month++){
		if (isleapyear(t.year)) n += nSecondsInMonths_Leap[month];
		else n += nSecondsInMonths_NonLeap[month];
	}
	n += (t.day - 1) * 86400;
	n+= t.hour * 3600;
	n += t.minute * 60;
	n += t.second;
	return n;
}
void gettime(time *buff,long int s)
{
	int year,month,n;
	year=INIT_YEAR;
	n=isleapyear(year)?nSecondsInLeapYear:nSecondsInNonLeapYear;
	while(s>=n){
		s-=n;
		year++;
		n=isleapyear(year)?nSecondsInLeapYear:nSecondsInNonLeapYear;
	}
	month=1;
	n=isleapyear(year)?nSecondsInMonths_Leap[month]:nSecondsInMonths_NonLeap[month];
	while(s>=n){
		s-=n;
		month++;
		n=isleapyear(year)?nSecondsInMonths_Leap[month]:nSecondsInMonths_NonLeap[month];
	}
	buff->year=year;
	buff->month=month;
	buff->day=s/86400+1;s=s%86400;
	buff->hour=s/3600;s=s%3600;
	buff->minute=s/60;s=s%60;
	buff->second=s;
}
void getcountdown(time *buff,long int s)
{
	buff->year=s/nSecondsInNonLeapYear;s=s%nSecondsInNonLeapYear;
	buff->month=s/nSecondsInApril;s=s%nSecondsInApril;
	buff->day=s/86400+1;s=s%86400;
	buff->hour=s/3600;s=s%3600;
	buff->minute=s/60;s=s%60;
	buff->second=s;
}
int writetime(char *buff,time t,TIMEFORMAT f)
{
	switch(f){
	case DATE_AND_TIME:
		strprintf(buff,"%04d.%02d.%02d %02d:%02d:%02d",t.year,t.month,t.day,t.hour,t.minute,t.second);
	return 19;
	case TIME:
		strprintf(buff,"%02d:%02d:%02d",t.hour,t.minute,t.second);
	return 8;
	case COUNTDOWN:
		if(t.year>0){strprintf(buff,"%04dy%02dm%02dd %02d:%02d:%02d",t.year,t.month,t.day,t.hour,t.minute,t.second);return 20;}
		if(t.month>0){strprintf(buff,"%02dm%02dd %02d:%02d:%02d",t.month,t.day,t.hour,t.minute,t.second);return 15;}
		if(t.day>0){strprintf(buff,"%02dd %02d:%02d:%02d",t.day,t.hour,t.minute,t.second);return 12;}
		if(t.hour>0){strprintf(buff,"%02d:%02d:%02d",t.hour,t.minute,t.second);return 8;}
		if(t.minute>0 || t.second>0){strprintf(buff,"%02d:%02d",t.minute,t.second);return 5;}
		strprintf(buff,"--:--");return 5;
	}
}
void readtime(time *buff)
{
	buff->year=RTCReadYear();
	buff->month=RTCReadMonth();
	buff->day=RTCReadDayOfMonth();
	buff->hour=RTCReadHour();
	buff->minute=RTCReadMinute();
	buff->second=RTCReadSecond();
}

#define FONT_BIG_HEIGHT 7

typedef enum {BIG,SMALL} FONT;

#define MAX_CLOCK_NUMBER 5

typedef struct{
	int x;
	int y;
	FONT font;
	TIMEFORMAT format;
	long int countdown;
} clock;
clock clockarray[MAX_CLOCK_NUMBER];
int nclock;
int alarm;
void clockevent()
{
	char buff[30];
	time t;
	int i,j,x;

	for(i=0;i<nclock;i++){
		switch(clockarray[i].format){
		case DATE_AND_TIME:
		case TIME:
			readtime(&t);
			writetime(buff,t,clockarray[i].format);
			x=clockarray[i].x;
		break;
		case COUNTDOWN:
			j=clockarray[i].countdown;
			if(j==0)alarm=1;
			getcountdown(&t,j);
			writetime(buff,t,clockarray[i].format);
			clockarray[i].countdown--;
		break;
		}
		if(clockarray[i].font==BIG)PrintXY(x,clockarray[i].y,buff,0);
		else PrintMini(x,clockarray[i].y,buff,0);
	}
	Bdisp_PutDisp_DD();
}
int addclock(int x,int y,FONT f,TIMEFORMAT tf,long int countdown)
{
	if(nclock>=MAX_CLOCK_NUMBER) return 1;
	clockarray[nclock].x=x;
	clockarray[nclock].y=y;
	clockarray[nclock].font=f;
	clockarray[nclock].format=tf;
	clockarray[nclock].countdown=countdown;
	nclock++;
	return 0;
}
void clearclock()
{
	nclock=0;
}

	alarm=0;

int ui_textbox(char *buffer,int *init_length,int max_length,int x,int y,FONT font,int width,int password)
{
	unsigned int key;
	int i,ischar,height,length;
	char buff[30],puff[30];
	char *outbuff;
	DISPBOX area;

	if(width==0 && font==BIG) width=max_length*(FONT_BIG_WIDTH+1);
	if(font==BIG) height=FONT_BIG_HEIGHT; else height=FONT_SMALL_HEIGHT;
	area.left=x;area.top=y;area.right=x+width-1;area.bottom=y+height-1;
	if(password) outbuff=puff; else outbuff=buff;
	length=*init_length;
	for(i=0;i<length;i++){
		buff[i]=buffer[i];puff[i]='*';
	}
	buff[length]=0;puff[length]=0;
	key=ischar=0;
	while(key!=KEY_CTRL_EXE && key!=KEY_CTRL_EXIT && key!=KEY_CTRL_UP && key!= KEY_CTRL_DOWN){
		if(length<max_length){
			buff[length]='_';puff[length]='_';
			buff[length+1]=0;puff[length+1]=0;
		}else{
			buff[length]=0;puff[length]=0;
		}
		if(font==BIG)PrintXY(x,y,outbuff,0);
		else PrintMini(x,y,outbuff,MINI_OVER);
		ischar=GetKey(&key);
		Bdisp_AreaClr_DDVRAM(&area);

		if(key==KEY_CTRL_AC) length=0;
		if(key==KEY_CTRL_DEL && length>0) length--;
		if(ischar && isalnum(key) && length<max_length){
			buff[length]=key;puff[length]='*';length++;
		}
	}

	if(key!=KEY_CTRL_EXIT){
		for(i=0;i<length;i++) buffer[i]=buff[i];
		buffer[length]=0;buff[length]=0;puff[length]=0;
		*init_length=length;
	}else{
		length=*init_length;
		for(i=0;i<length;i++){
			buff[i]=buffer[i];puff[i]='*';
		}
		buff[length]=0;puff[length]=0;
	}
	if(font==BIG)PrintXY(x,y,outbuff,0);
	else PrintMini(x,y,outbuff,MINI_OVER);
	switch(key){
		case KEY_CTRL_EXIT:
			return 4;
		case KEY_CTRL_UP:
			return -1;
		case KEY_CTRL_EXE:
		case KEY_CTRL_DOWN:
			return 1;
	}
}
